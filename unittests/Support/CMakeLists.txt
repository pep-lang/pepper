cmake_minimum_required(VERSION 3.14)

set(CMAKE_CXX_STANDARD 20)

set(SRC_FILES Debug.cpp)

add_library(TestSupport SHARED ${SRC_FILES})

if (MSVC)
    set(TEST_SUPPORT_SHARED_LIBRARIES ${CMAKE_CURRENT_BINARY_DIR}/TestSupport.lib)
    set(TEST_SUPPORT_RUNTIME_LIBRARIES ${CMAKE_CURRENT_BINARY_DIR}/TestSupport.dll PARENT_SCOPE)
elseif (APPLE)
    set(TEST_SUPPORT_SHARED_LIBRARIES ${CMAKE_CURRENT_BINARY_DIR}/TestSupport.dylib)
else ()
    set(TEST_SUPPORT_SHARED_LIBRARIES ${CMAKE_CURRENT_BINARY_DIR}/TestSupport.so)
endif ()

set(TEST_SUPPORT_INCLUDE_DIRS . PARENT_SCOPE)
set(TEST_SUPPORT_LIBRARIES ${TEST_SUPPORT_SHARED_LIBRARIES} PARENT_SCOPE)
