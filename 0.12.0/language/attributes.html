The Pepper Programming Language

Attributes

An attribute is a name in square brackets that tells the compiler something about the declaration below it. Each attribute goes on its own line, above its target, as pep-fmt writes it, except a parameter's, which stay on its line, in front of it:

# Attributes, each on its own line above what it applies to.

[Target(OS(Windows))]
system_name(): String {
    return "Windows";
}

[Target(!OS(Windows))]
system_name(): String {
    return "not Windows";
}

[Deprecated("use 'system_name' instead", "system_name")]
os_name(): String {
    return system_name();
}

# A parameter's attributes stay on its line, in front of it.
keep([Move] names: Array<String>): ulong {
    return names.count;
}

main(): int {
    print(system_name().size > 0);
    var names = ["a", "b"];
    print(keep(move(names)));
    return 0;
}

# Output:
# true
# 2
